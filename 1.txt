
function mydebounce(func, ms) {
  let timeout;
  return function (...args) {
    clearTimeout(timeout);
    timeout = setTimeout(() => {
      func.apply(this, args);
    }, ms);
  };
}
export const Layout = () => {
  const [userName, setUserName] = useState("");
  const [showUserName, setShowUserName] = useState("");

  function handleName(name) {
    setUserName(name);
    const debounceUserName = mydebounce(setShowUserName, 5000);
    debounceUserName(name);
  }

  return (
    <div>
      <Input username={userName} setUserName={handleName} placeholder="Введите имя" />
      <div className="text">Поиск по {showUserName}</div>;
      <div>
        <Table />
      </div>
    </div>
  );
};





import { useMemo, useState } from "react";
import { Input } from "../Input";
import { Table } from "../Table";

function mydebounce(func, ms) {
  let timeout;
  return function (...args) {
    clearTimeout(timeout);
    timeout = setTimeout(() => {
      func.apply(this, args);
    }, ms);
  };
}

export const Layout = () => {
  const [userName, setUserName] = useState("");
  const [showUserName, setShowUserName] = useState("");
  
  const debounceUserName = useMemo(() => {
    return mydebounce(setShowUserName, 500);
  }, []);

  function handleName(name) {
    setUserName(name);
    debounceUserName(name);
  }

  return (
    <div>
      <Input username={userName} setUserName={handleName} placeholder="Введите имя" />
      <div className="text">Поиск по {showUserName}</div>
      <div>
        <Table />
      </div>
    </div>
  );
};
